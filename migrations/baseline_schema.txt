  column id: INTEGER, nullable=False, default=nextval('audit_log_id_seq'::regclass)
  column question: TEXT, nullable=False, default=None
  column reason: TEXT, nullable=True, default=None
  column sources: TEXT, nullable=True, default=None
  column tier: VARCHAR(10), nullable=False, default=None
  column timestamp: TIMESTAMP WITHOUT TIME ZONE, nullable=True, default=None
  column top_distance: DOUBLE PRECISION, nullable=True, default=None
  column warnings: TEXT, nullable=True, default=None
  primary key: ['id']
